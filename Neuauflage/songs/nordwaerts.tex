\begin{song}{Nordwärts, nordwärts}{Volkslied}{Capo -}

\cl{  Am                   G                Dm            Am}
\slbl{1.}Nordwärts, nordwärts wolln wir ziehen zu den Bergen und den Seen,
\cl{C            G             Am        Dm  Em  Am}
wollen neues Land erleben, wolln auf Fahrten geh'n.

\cl{  Am              G              Dm            Am}
\slbl{2.}Wollen frei, so wie ein Vogel, wiegen uns im kalten Wind,
\cl{C                 G              Am       Dm   Em   Am}
wolln den Ruf der Wildnis hören, wenn wir glücklich sind.

\cl{  Am                   G            Dm           Am}
\slbl{3.}Wolln durch Moor und Sümpfe waten abends legen uns zur Ruh,
\cl{C               G               Am     Dm Em  Am}
Klampfen sollen leis erklingen, singen im-mer zu.

\cl{  Am           G                 Dm             Am}
\slbl{4.}In der Kohte brennt ein Feuer, füllt uns alle mit Bedacht.
\cl{C                     G              Am       Dm  Em  Am}
Schlaf senkt sich auf uns hernieder, doch die Wildnis wacht.

\cl{  Am                G              Dm             Am}
\slbl{5.}Käutzchenschreie, Bäume rauschen bis zum frühen Morgengrau.
\cl{C         G              Am          Dm Em  Am}
Über ausgequalmten Feuer strahlt der Himmel blau.

\cl{  Am              G                 Dm           Am}
\slbl{6.}Wenn wir wieder Heimwärts ziehen, sehnet jeder sich zurück.
\cl{C               G                Am    Dm  Em  Am}
Denkt an die vergangnen Fahrten, an vergangnes Glück.

\cl{  Am                   G                 Dm            Am}
\slbl{7.}Nordwärts, nordwärts wolln wir wieder, zu den Bergen und den Seen,
\cl{C               G           Am      Dm  Em  Am}
dieses Land nochmal erleben und auf Fahrten gehn.



\griffe{Dm}






\schlagmuster{standard}[Ganzes Lied]
\end{song}
